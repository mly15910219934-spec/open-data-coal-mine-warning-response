from pathlib import Path
import zipfile, json
from xml.etree import ElementTree as ET

ROOT = Path(__file__).resolve().parents[1]
source = Path('AUTHOR_LOCAL_PATH_REQUIRED')
NS = {'w': 'http://schemas.openxmlformats.org/wordprocessingml/2006/main'}
with zipfile.ZipFile(source) as z:
    xml = ET.fromstring(z.read('word/document.xml'))
tables = []
for i, t in enumerate(xml.findall('.//w:tbl', NS), 1):
    rows = []
    for row in t.findall('w:tr', NS):
        rows.append([''.join(x.text or '' for x in c.findall('.//w:t', NS)) for c in row.findall('w:tc', NS)])
    tables.append({'table_index': i, 'rows': rows})
paras = [''.join(x.text or '' for x in p.findall('.//w:t', NS)) for p in xml.findall('.//w:p', NS)]
out = {'source': str(source), 'tables': tables, 'captions': [p for p in paras if p.startswith(('Table ', 'Fig ', 'S1 Fig'))]}
(ROOT/'logs/manuscript_baseline_tables.json').write_text(json.dumps(out, indent=2, ensure_ascii=False), encoding='utf-8')
print(json.dumps(out, indent=2, ensure_ascii=False))
