def verify_pre(pipe,name,xf):
    cfg,_=old.configs();pre=pipe.named_steps['preprocess']
    cats=pre.named_transformers_['categorical'].categories_
    for c,seen in zip(cfg['categorical_features'],cats):assert set(seen)==set(xf[c])
    if name in ['Logistic Regression','SVM']:
        sc=pre.named_transformers_['numeric'];assert int(sc.n_samples_seen_)==len(xf)
        assert np.allclose(sc.mean_,xf[cfg['numeric_features']].mean().to_numpy())
    return {'categorical_fit_verified':True,'numeric_fit_verified':True,'preprocessing_fit_n':len(xf)}
def ros_model(base):
    from imblearn.pipeline import Pipeline
    from imblearn.over_sampling import RandomOverSampler
    est=clone(base.named_steps['model'])
    if 'class_weight' in est.get_params():est.set_params(class_weight=None)
    return Pipeline([('preprocess',clone(base.named_steps['preprocess'])),('resample',RandomOverSampler(random_state=SEED)),('model',est)])
def cv_run(ros=False):
    guard();Xtr,Xte,ytr,yte,X,y=data();models=old.build_models();rows=[];audit=[]
    split=RepeatedStratifiedKFold(n_splits=5,n_repeats=10,random_state=SEED)
    for i,(ti,vi) in enumerate(split.split(Xtr,ytr)):
        xf,yf=Xtr.iloc[ti],ytr.iloc[ti];xv,yv=Xtr.iloc[vi],ytr.iloc[vi]
        assert set(xf.index).isdisjoint(xv.index) and set(xf.index).isdisjoint(Xte.index) and set(xv.index).isdisjoint(Xte.index)
        entry={'Repeat':i//5+1,'Fold':i%5+1,'train_indices':xf.index.tolist(),'validation_indices':xv.index.tolist(),'train_positive':int(yf.sum()),'train_negative':int((yf==0).sum()),'fits':[]}
        for name in NAMES:
            pipe=ros_model(models[name]) if ros else clone(models[name])
            pipe.fit(xf,yf) if ros else old.fit(pipe,name,xf,yf)
            check=verify_pre(pipe,name,xf);check['Model']=name
            if ros:
                ids=pipe.named_steps['resample'].sample_indices_;assert ids.min()>=0 and ids.max()<len(xf)
                resampled=yf.iloc[ids];assert int(resampled.sum())==int((resampled==0).sum())
                check.update({'resampled_n':len(ids),'resampled_positive':int(resampled.sum()),'resampled_original_indices':xf.index[ids].tolist(),'class_weight':None,'sample_weight':'none'})
            else:
                check.update({'class_weight':'balanced' if name!='Gradient Boosting' else None,'sample_weight':'balanced' if name=='Gradient Boosting' else 'none','training_weight_negative':len(yf)/(2*sum(yf==0)),'training_weight_positive':len(yf)/(2*sum(yf==1))})
            entry['fits'].append(check);s=pipe.predict_proba(xv)[:,1]
            rows.append({'Model':name,'Strategy':'RandomOverSampler' if ros else 'Balanced weighting','Repeat':i//5+1,'Fold':i%5+1,'ROC-AUC':roc_auc_score(yv,s),'PR-AUC':average_precision_score(yv,s),'Positive count':int(yv.sum()),'Negative count':int((yv==0).sum()),'Validation n':len(yv)})
        audit.append(entry)
        print(('ROS' if ros else 'MAIN')+f' repeated CV {i+1}/50 complete',flush=True)
    df=write('resampling_cv_fold_results.csv' if ros else 'repeated_cv_fold_results.csv',rows)
    summary=[]
    for name in NAMES:
        d=df[df.Model==name];r={'Model':name,'Strategy':'RandomOverSampler' if ros else 'Balanced weighting','Validation folds':len(d)}
        for metric in ['ROC-AUC','PR-AUC']:r[metric+' mean']=d[metric].mean();r[metric+' SD']=d[metric].std(ddof=1)
        summary.append(r)
    write('resampling_cv_summary.csv' if ros else 'repeated_cv_summary.csv',summary)
    dump(OUT/('resampling_cv_fit_audit.json' if ros else 'repeated_cv_fit_audit.json'),audit);guard()
def bootstrap():
    guard();p=read('held_out_predictions.csv');y=p.label.to_numpy();perf=read('model_performance_revision.csv').set_index('Model')
    metrics=['ROC-AUC','PR-AUC','Recall','Precision','Brier Score'];rows=[];receipts=[];rng=np.random.default_rng(SEED)
    for b in range(2000):
        idx=np.r_[rng.choice(np.flatnonzero(y==1),sum(y==1),replace=True),rng.choice(np.flatnonzero(y==0),sum(y==0),replace=True)]
        receipts.append(idx.tolist())
        for name in NAMES:
            s=p[ABBR[name]+'_score'].to_numpy()[idx];r=old.score_row(name,y[idx],s)
            rows.append({'Resample':b+1,'Model':name,**{k:r[k] for k in metrics},'Positive count':int(sum(y[idx]==1)),'Negative count':int(sum(y[idx]==0)),'Precision zero division applied':bool(sum(s>=.5)==0),'Undefined metric reason':'No positive predictions; precision defined as zero' if sum(s>=.5)==0 else 'none'})
        if (b+1)%500==0:print('Bootstrap',b+1,'/2000',flush=True)
    df=write('bootstrap_2000_raw.csv',rows);summary=[]
    for name in NAMES:
        point=old.score_row(name,y,p[ABBR[name]+'_score'].to_numpy());d=df[df.Model==name]
        for metric in metrics:
            assert point[metric]==perf.loc[name,metric],('Point estimate mismatch',name,metric)
            lo,hi=np.quantile(d[metric],[.025,.975])
            summary.append({'Model':name,'Metric':metric,'Point estimate':point[metric],'Lower 95% CI':lo,'Upper 95% CI':hi,'Resamples':len(d),'Zero division convention count':int(d['Precision zero division applied'].sum()) if metric=='Precision' else 0})
    write('bootstrap_2000_summary.csv',summary)
    np.savez_compressed(OUT/'bootstrap_paired_indices.npz',indices=np.array(receipts,dtype=np.int32))
    guard()
