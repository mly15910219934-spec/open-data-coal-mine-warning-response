def load(n):

def md(headers,rows):return '\n'.join(['| '+' | '.join(headers)+' |','| '+' | '.join(['---']*len(headers))+' |']+['| '+' | '.join(map(str,r))+' |' for r in rows])

def doc(title,landscape=False):
    d=Document();s=d.sections[0];s.page_width=Inches(11.69 if landscape else 8.27);s.page_height=Inches(8.27 if landscape else 11.69)
    s.top_margin=s.bottom_margin=s.left_margin=s.right_margin=Inches(.65)
    for st in d.styles:
        for el in list(st.element.iter(qn('w:pBdr'))):el.getparent().remove(el)
    for name,size in [('Normal',10.5),('Title',15),('Heading 1',12),('Heading 2',11)]:
        st=d.styles[name];st.font.name='Arial';st.font.size=Pt(size);st.font.color.rgb=RGBColor(0,0,0)
    d.styles['Normal'].paragraph_format.space_after=Pt(6);d.styles['Normal'].paragraph_format.line_spacing=1.08
    d.styles['Heading 1'].paragraph_format.space_before=Pt(12);d.styles['Heading 1'].paragraph_format.space_after=Pt(4)
    d.add_paragraph(title,'Title');return d
def table(d,headers,rows,widths):
    t=d.add_table(rows=1,cols=len(headers));t.autofit=False
    for c,w in zip(t.columns,widths):c.width=Inches(w)
    for c,h in zip(t.rows[0].cells,headers):c.text=h
    for r in rows:
        for c,v in zip(t.add_row().cells,r):c.text=str(v)
    borders=OxmlElement('w:tblBorders')
    for edge in ['top','left','bottom','right','insideH','insideV']:
        e=OxmlElement('w:'+edge);e.set(qn('w:val'),'single');e.set(qn('w:sz'),'4');e.set(qn('w:color'),'D9D9D9');borders.append(e)
    t._tbl.tblPr.append(borders)
    for i,row in enumerate(t.rows):
        for c,w in zip(row.cells,widths):
            c.width=Inches(w);c.vertical_alignment=1
            mar=OxmlElement('w:tcMar')
            for edge in ['top','bottom','left','right']:
                e=OxmlElement('w:'+edge);e.set(qn('w:w'),'80');e.set(qn('w:type'),'dxa');mar.append(e)
            c._tc.get_or_add_tcPr().append(mar)
            if i==0:e=OxmlElement('w:shd');e.set(qn('w:fill'),'E8EDF2');c._tc.get_or_add_tcPr().append(e)
            for p in c.paragraphs:
                p.alignment=WD_ALIGN_PARAGRAPH.CENTER;p.paragraph_format.space_after=Pt(3);p.paragraph_format.space_before=Pt(3)
                for run in p.runs:run.font.size=Pt(9);run.bold=i==0
        row._tr.get_or_add_trPr().append(OxmlElement('w:cantSplit'))
    t.rows[0]._tr.get_or_add_trPr().append(OxmlElement('w:tblHeader'));d.add_paragraph()

def sheet(name,file):
    rows=load(file);headers=list(rows[0]);matrix=[]
    for r in rows:
        rr=[]
        for k in headers:
            v=r[k]
            if k in ['All tied cutoffs','Interpretation','Scenario','Undefined metric reason']:pass
            elif v in ['True','False']:v=v=='True'
            else:
                try:v=float(v);v=int(v) if v.is_integer() else v
                except (ValueError,AttributeError):pass
            rr.append(v)
        matrix.append(rr)
    display_names={'Warning workload':'Held-out test-set alert burden','Warning workload percentage':'Held-out test-set alert burden percentage','Threshold':'Decision cut-off','Decision cutoff':'Decision cut-off'}
    return {'name':name,'source':'results/'+file,'headers':[display_names.get(h,h) for h in headers],'rows':matrix}
