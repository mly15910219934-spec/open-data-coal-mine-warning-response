from __future__ import annotations

import io
import urllib.request
import zipfile
from datetime import date

from .utils import ROOT, ensure_output_dirs, load_yaml, sha256, write_json


def main() -> None:
    ensure_output_dirs()
    cfg = load_yaml("config/frozen_analysis_v2.yaml")["dataset"]
    archive = ROOT / "data/raw" / cfg["archive_file"]
    with urllib.request.urlopen(cfg["url"]) as response:
        archive.write_bytes(response.read())
    with zipfile.ZipFile(io.BytesIO(archive.read_bytes())) as bundle:
        member = next(name for name in bundle.namelist() if name.endswith(cfg["data_file"]))
        target = ROOT / "data/raw" / cfg["data_file"]
        target.write_bytes(bundle.read(member))
    metadata = {"dataset": cfg["name"], "doi": cfg["doi"], "source_url": cfg["url"], "download_date": date.today().isoformat(), "archive_file": archive.name, "archive_sha256": sha256(archive), "data_file": target.name, "data_sha256": sha256(target)}
    write_json(ROOT / "outputs/metadata/data_access.json", metadata)
    print(target)


if __name__ == "__main__":
    main()
