"""Render the saved revision results; does not retrain or alter scores."""
from pathlib import Path
import shutil, json
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from PIL import Image

ROOT=Path(__file__).resolve().parents[1]
R=ROOT/'results'; F=ROOT/'figures'
NAMES=['Logistic Regression','Random Forest','Gradient Boosting','SVM']
COLORS=dict(zip(NAMES,['#1f77b4','#ff7f0e','#2ca02c','#d62728']))
SHORT=dict(zip(NAMES,['LR','RF','GB','SVM']))
plt.rcParams.update({'font.family':'Arial','font.size':10,'axes.labelsize':11,
    'legend.fontsize':9,'lines.linewidth':1.7,'savefig.facecolor':'white',
    'svg.fonttype':'none','axes.spines.top':False,'axes.spines.right':False})
qc=[]
def export(fig,stem):
    fig.tight_layout(pad=1.3)
    fig.savefig(F/(stem+'.png'),dpi=300)
    fig.savefig(F/(stem+'.svg'))
    with Image.open(F/(stem+'.png')) as im:
        rgb=Image.new('RGB',im.size,'white')
        if 'A' in im.getbands():rgb.paste(im.convert('RGBA'),mask=im.getchannel('A'))
        else:rgb=im.convert('RGB')
        rgb.save(F/(stem+'.tif'),compression='tiff_lzw',dpi=(300,300))
    plt.close(fig)
    with Image.open(F/(stem+'.tif')) as im:
        im.load(); size=(F/(stem+'.tif')).stat().st_size
        ok=im.format=='TIFF' and im.mode=='RGB' and im.info.get('compression')=='tiff_lzw' and min(im.info['dpi'])>=300 and size<10*1024**2
        qc.append({'File':stem+'.tif','Pixels':str(im.size),'DPI':str(im.info['dpi']),'Mode':im.mode,'Compression':im.info['compression'],'Size MB':size/1024**2,'PASS':ok})
        assert ok

perf=pd.read_csv(R/'model_performance_revision.csv').set_index('Model')
curves=pd.read_csv(R/'curve_source_data.csv')
for kind,stem in [('ROC','Fig4_revision'),('PR','Fig5_revision')]:
    fig,ax=plt.subplots(figsize=(7.5,5))
    for name in NAMES:
        q=curves[(curves.Model==name)&(curves.Curve==kind)]
        value=perf.loc[name,'ROC-AUC' if kind=='ROC' else 'PR-AUC']
        ax.plot(q.x,q.y,color=COLORS[name],label=f'{name} ({"AUC" if kind=="ROC" else "AP"} = {value:.3f})',drawstyle='default' if kind=='ROC' else 'steps-post')
    if kind=='ROC':
        ax.plot([0,1],[0,1],'--',color='0.5',lw=1);ax.set(xlabel='False positive rate',ylabel='True positive rate');ax.legend(loc='lower right')
    else:
        prev=51/776;ax.axhline(prev,ls='--',color='0.5',lw=1,label=f'Test-set prevalence ({prev:.3f})');ax.set(xlabel='Recall',ylabel='Precision');ax.legend(loc='upper right')
    ax.set(xlim=(0,1),ylim=(0,1.02));export(fig,stem)

bins=pd.read_csv(R/'calibration_bin_data.csv')
fig,axes=plt.subplots(2,2,figsize=(7.5,6.5),sharex=True,sharey=True)
for ax,name,letter in zip(axes.flat,NAMES,'abcd'):
    q=bins[bins.Model==name]
    ax.plot([0,1],[0,1],'--',color='0.5',lw=1)
    ax.plot(q['Mean decision score'],q['Observed positive fraction'],'o-',color=COLORS[name],ms=4)
    ax.set(title=f'({letter}) {SHORT[name]}  Brier = {perf.loc[name,"Brier Score"]:.3f}',xlim=(0,1),ylim=(0,1))
for ax in axes[1,:]:ax.set_xlabel('Mean model decision score')
for ax in axes[:,0]:ax.set_ylabel('Observed positive fraction')
export(fig,'Fig6_revision')
for ext in ['png','tif']:shutil.copy2(F/('Fig6_revision.'+ext),F/('calibration_curve_revision.'+ext))

sweep=pd.read_csv(R/'threshold_analysis_all_models.csv')
fig,axes=plt.subplots(2,1,figsize=(7.5,6.5),sharex=True)
for name in NAMES:
    q=sweep[sweep.Model==name]
    axes[0].plot(q.Threshold,q.Recall,color=COLORS[name],label=name)
    axes[1].plot(q.Threshold,q['Warning workload'],color=COLORS[name])
axes[0].set(ylabel='Recall',ylim=(0,1.03),title='(a) Hazardous cases identified');axes[0].legend(loc='upper right')
axes[1].set(xlabel='Decision cut-off',ylabel='Held-out test-set alerts',ylim=(0,800),xlim=(0,1),title='(b) Test-set alert burden')
export(fig,'Fig7_revision')

matched=pd.read_csv(R/'matched_alert_burden_analysis.csv')
fig,ax=plt.subplots(figsize=(7.5,5))
for name in NAMES:
    q=matched[matched.Model==name]
    ax.plot(q['Warning workload'],q.Recall,'o-',color=COLORS[name],label=name,ms=5)
ax.set(xlabel='Actual held-out test-set alerts',ylabel='Recall',ylim=(0,1.03));ax.legend(loc='lower right')
export(fig,'alert_burden_recall')
pd.DataFrame(qc).to_csv(R/'figure_quality_checks.csv',index=False)
(ROOT/'reports/figure_qc_report.md').write_text('# Figure quality checks\n\n'+pd.DataFrame(qc).to_string(index=False)+'\n\nEach PNG was rendered at its final pixel dimensions. TIFFs are RGB with LZW compression. SVG files retain editable text. Calibration aliases contain the same figure as Fig6.\n',encoding='utf-8')
print(json.dumps(qc,indent=2))
